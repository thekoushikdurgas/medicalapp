# ==============================================================================
# 4. DATABASE & ALEMBIC MIGRATIONS
# ==============================================================================

# Navigate to backend directory before running Alembic
cd backend

# Create a new auto-detected migration revision
alembic revision --autogenerate -m "add_fefo_and_purchase_entry_models"

# Apply all pending migrations to the latest database schema
alembic upgrade head

# Roll back the single most recent migration
alembic downgrade -1

# Roll back all migrations to initial state
alembic downgrade base

# View migration history and current head
alembic history --verbose
alembic current

# Connect directly to local PostgreSQL instance
psql -h localhost -p 5432 -U postgres -d pharmacyfantasy

# Useful PostgreSQL commands inside psql:
#   \dt                  -- List all relational tables
#   \d+ stock_batches    -- Inspect StockBatch schema, columns, and indexes
#   \d+ orders           -- Inspect Order schema and foreign keys
#   SELECT count(*) FROM products;
#   SELECT batch_number, expiry_date, quantity_available FROM stock_batches ORDER BY expiry_date ASC;

# 12. TROUBLESHOOTING & COMMON ERROR FIXES
# ==============================================================================

# Issue: Port 8000 or 3000 already in use (Windows)
netstat -ano | findstr :8000
netstat -ano | findstr :3000
taskkill /F /PID <PID_NUMBER>

# Issue: PostgreSQL connection refused
docker compose restart postgres
docker compose logs --tail=50 postgres

# Issue: Redis connection failure
docker compose restart redis

# Issue: Alembic migration divergence / multiple heads
alembic merge heads -m "merge_divergent_heads"
alembic upgrade head

# Issue: Next.js script execution policy error in PowerShell
# Run commands using cmd wrapper:
cmd /c npm run dev
cmd /c npm run build
cmd /c npx tsc --noEmit
